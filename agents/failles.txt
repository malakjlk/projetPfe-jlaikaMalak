import os
import subprocess
from typing import Optional

def generer_rapport(commande: str) -> Optional[str]:
    """
    Exécute l'outil ``report_tool`` avec la commande fournie et renvoie la sortie
    texte du processus.  Aucun appel à un shell n'est effectué afin d'éviter
    toute injection de commande (CWE‑78).

    Args:
        commande: Chaîne représentant la sous‑commande à passer à ``report_tool``.
                  Elle est traitée comme un argument unique et ne subit aucune
                  interpolation.

    Returns:
        Le texte de sortie du processus si l'exécution réussit, sinon ``None``.
        Les erreurs d'exécution (ex. outil introuvable, timeout) sont capturées
        et renvoient ``None`` plutôt que de lever d'exception métier, afin de
        respecter le contrat de la fonction PHP d'origine qui renvoie ``null``
        en cas d'échec.
    """
    # Validation très basique : la commande ne doit pas contenir de caractères
    # de contrôle ou de séparateurs de commande.  On accepte uniquement les
    # caractères alphanumériques, les tirets, les underscores et les points.
    if not commande or any(c for c in commande if not (c.isalnum() or c in "-_.")):
        # En cas de valeur suspecte, on renvoie None comme le PHP le ferait
        # (pas d'exception levée ici).
        return None

    try:
        result = subprocess.run(
            ["report_tool", commande],
            capture_output=True,
            text=True,
            timeout=30,
            shell=False,          # impératif pour éviter l'injection de commande
        )
        # Si le processus s'est terminé avec un code de retour non‑zéro,
        # on considère que la génération a échoué et on renvoie None.
        if result.returncode != 0:
            return None
        return result.stdout
    except (OSError, subprocess.SubprocessError):
        # Erreurs d'exécution (ex. binaire absent, timeout, etc.)
        return None

from typing import List, Dict, Optional
from sqlalchemy.orm import Session
from sqlalchemy import text


def chercher_produit(nom: str, db: Session) -> Optional[List[Dict]]:
    """
    Recherche les produits dont le nom contient la chaîne fournie.

    Args:
        nom: Chaîne à rechercher dans la colonne ``nom`` de la table ``produits``.
        db:  Session SQLAlchemy déjà configurée (injection de dépendance dans FastAPI).

    Returns:
        - Une liste de dictionnaires, chaque dict contenant les colonnes du produit
          (ex. result[0]["id"], result[0]["nom"], …) si au moins un enregistrement est trouvé.
        - ``None`` si aucune ligne ne correspond à la recherche.
    """
    # Utilisation d'une requête paramétrée pour éviter toute injection SQL (CWE‑89)
    stmt = text(
        "SELECT * FROM produits WHERE nom LIKE :pattern"
    )
    # Le pattern inclut les jokers % autour du terme recherché
    result = db.execute(stmt, {"pattern": f"%{nom}%"})

    # ``mappings()`` renvoie chaque ligne sous forme de dict‑like
    rows = result.mappings().all()

    return rows if rows else None
